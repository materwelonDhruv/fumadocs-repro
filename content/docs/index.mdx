title: Repro
---

This page has ten `twoslash` code blocks with `^?` queries on deep zod types.
Serving it under `next dev` exhausts the heap and crashes; see the README.

```ts twoslash
import { z } from 'zod';
const a = z.object({
  id: z.string(),
  n: z.number(),
  tags: z.array(z.string()),
  meta: z.object({ a: z.string(), b: z.number(), c: z.boolean() }),
});
type A = z.infer<typeof a>;
//   ^?
```

```ts twoslash
import { z } from 'zod';
const b = z.object({
  user: z.object({ name: z.string(), roles: z.array(z.enum(['a', 'b', 'c'])) }),
  flags: z.record(z.string(), z.boolean()),
});
type B = z.infer<typeof b>;
//   ^?
```

```ts twoslash
import { z } from 'zod';
const c = z.discriminatedUnion('kind', [
  z.object({ kind: z.literal('x'), x: z.number() }),
  z.object({ kind: z.literal('y'), y: z.string() }),
]);
type C = z.infer<typeof c>;
//   ^?
```
